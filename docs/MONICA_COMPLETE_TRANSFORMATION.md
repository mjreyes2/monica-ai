## 📖 Usage Examples

### Complete Session Example
```python
from monica_knowledge_system import MonicaExpert, MonicaKnowledgeBase

# Initialize Monica
knowledge = MonicaKnowledgeBase()
monica = MonicaExpert(knowledge)

# === PERSONALITY ===
monica.set_accent("brooklyn")
monica.set_humor_style("ghetto")

# Tell a joke
joke = monica.tell_joke("quantum physics")
print(f"Monica: {joke}")

# === TEACHING ===
lesson = monica.teach(
    subject="computer science algorithms",
    level="beginner",
    person_id="user1"
)
print(f"\nLesson:\n{lesson}")

# === TRANSLATION ===
translation = monica.translate(
    "How do you become a computer scientist?",
    "english",
    "spanish"
)
print(f"\nSpanish: {translation['translation']}")

# === LEARNING ===
# Feed Monica a textbook
result = knowledge.learn_from_pdf("computer_science_textbook.pdf")
print(f"\nLearned from PDF: {result['pages']} pages")

# Feed her a lecture
result = knowledge.learn_from_audio("algorithms_lecture.mp3")
print(f"Learned from audio: {result['duration']} seconds")

# === MEMORY ===
# Remember user
knowledge.remember_person("user1", {
    "name": "Student",
    "preferences": {"accent": "brooklyn", "humor": "ghetto"},
    "topics": ["computer science", "quantum physics"],
    "note": "Wants to become a computer scientist"
})

# Recall later
person = knowledge.recall_person("user1")
print(f"\nI remember {person['name']}!")
print(f"Interactions: {person['interactions']}")
print(f"Topics: {', '.join(person['conversation_topics'])}")

# === THERAPY ===
# Provide trauma-informed support
response = monica.get_trauma_informed_response(
    user_state="stressed about CS degree",
    context="Feeling overwhelmed by algorithms exam"
)
print(f"\nMonica (supportive):\n{response}")

# === DISCUSSION ===
# Expert conversation
response = monica.discuss(
    topic="computer science",
    user_message="Explain time complexity and Big O notation",
    person_id="user1"
)
print(f"\nMonica explains:\n{response}")
```
